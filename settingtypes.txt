# Grudgelands — game settings shown in the main menu under "Games"
# (lua_api.md:198-201). Format per builtin/settingtypes.txt in the engine
# checkout (reference_projects/luanti):
# `name (Readable name) type type_args`, parts separated by exactly one space.
# Context annotations (`[server]` and friends) are valid only in builtin's own
# file and are INVALID here (builtin/settingtypes.txt:19-21).
#
# This file is read by the main menu, not by the server: each runtime default
# lives in the `core.settings:get_bool` call that reads its setting.

[Atmosphere]

#    Master switch for the server-side atmosphere layer: dynamic shadows,
#    bloom, exposure and saturation sent per player with player:set_lighting.
#    With this off, grug_core.set_atmosphere does nothing and joining players
#    are left at the engine's own lighting defaults — i.e. the look a player
#    gets without the game asking for anything. Turn it off to compare, or if
#    a client's post-processing pipeline misbehaves.
#    Render settings themselves stay the player's own (Settings > Graphics);
#    this switch only decides whether the game asks for anything at all.
grug_atmosphere_enabled (Enable Grudgelands atmosphere presets) bool true

#    Zone-driven moods on top of the preset layer above: one sky, fog and
#    lighting mood per race region (dwarf, human, elf, undead, orc, troll),
#    plus underground, open ocean, the shared Battlegrounds and the two
#    dragon islands. A globalstep re-checks each player's zone about every
#    two seconds and sends one lighting, sky and cloud packet only when the
#    mood actually changes; a standing player costs nothing.
#    With this off the moods stay available as manual /atmosphere presets,
#    but nobody is moved between them automatically.
#    Requires the master switch above.
grug_atmosphere_zones (Drive the atmosphere from the player's zone) bool true

grug_nametag_guard_background (Guard nametag background) string #00000040

#    Show a small green sprite bar above injured combat mobs and guards. The
#    existing per-viewer nametag range and hysteresis also control this bar.
grug_injured_mob_hp_bars (Show injured mob health bars) bool true

[Map generation]

#    Deterministic wide coast profiles ship enabled under Round 9's
#    20 percent soft emerge-time gate.
grug_mapgen_r9_coast_band_enabled (Enable wide coast profiles) bool true

[World preparation]

#    Prepare the bounded world surface before entry instead of only the six
#    starts. Full progress counts finished horizontal tiles; mountains can take
#    longer. Deep mining, ocean depths and arbitrary flight are not prepared. Read once on the first fresh-world boot; changing this later does
#    not change that world's mode. Normal shutdown saves progress for restart.
grug_prepare_full_world (Prepare full world before entry) bool false

# Multiplier for all non-player combat attack damage, including fixed DoTs,
# auras and boss ground attacks. Player/environmental damage is unchanged.
# Loaded at startup; restart after changing. 1.0 restores base attack damage.
grug_mob_damage_scale (Non-player attack damage multiplier) float 1.5 0 10
